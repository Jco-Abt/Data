# ============================================================
# 3. FUNCIONES DE CONVERSIÓN
# ============================================================

def convertir_decimal(valor):

    valor = str(valor).strip()

    if not valor:
        raise ValueError("Valor numérico vacío")

    try:
        numero = Decimal(valor)

    except InvalidOperation:
        raise ValueError(
            f"Valor numérico inválido: {valor}"
        )

    if not numero.is_finite():
        raise ValueError(
            f"Valor numérico no finito: {valor}"
        )

    # DECIMAL(30,19) admite hasta 11 dígitos enteros.
    if abs(numero) >= LIMITE_ENTEROS:
        raise ValueError(
            f"Excede el rango de DECIMAL(30,19): {valor}"
        )

    # Verificar que no se pierda precisión al limitar a 19 decimales.
    with localcontext() as contexto:

        contexto.prec = 60

        numero_ajustado = numero.quantize(
            UNIDAD_ESCALA
        )

        if numero != numero_ajustado:
            raise ValueError(
                f"Excede los 19 decimales permitidos: {valor}"
            )

    return numero_ajustado


def convertir_entero(valor, columna):

    numero = convertir_decimal(valor)

    if numero != numero.to_integral_value():
        raise ValueError(
            f"{columna} debe ser un número entero: {valor}"
        )

    resultado = int(numero)

    if columna == "BRICK_COD":

        if resultado < 0 or resultado > 2147483647:
            raise ValueError(
                f"BRICK_COD fuera del rango INT: {valor}"
            )

    elif columna == "IDPERIODO":

        if resultado < 1 or resultado > 60:
            raise ValueError(
                f"IDPERIODO debe estar entre 1 y 60: {valor}"
            )

    return resultado


# ============================================================
# 4. CONSULTAR TIPO DE CAMBIO CORPORATIVO
# ============================================================

def obtener_tipo_cambio(cursor):

    anio = AUDIT_DATE.year

    cursor.execute(
        """
        SELECT TIPO_CAMBIO
        FROM dbo.Config_TipoCambio
        WHERE ANIO = ?
        """,
        anio
    )

    resultado = cursor.fetchone()

    if resultado is None:
        raise ValueError(
            f"No existe tipo de cambio para el año {anio}"
        )

    tipo_cambio = Decimal(str(resultado[0]))

    if not tipo_cambio.is_finite() or tipo_cambio <= 0:
        raise ValueError(
            f"Tipo de cambio inválido para {anio}"
        )

    return tipo_cambio


# ============================================================
# 5. VALIDAR QUE NO EXISTA UNA CARGA ANTERIOR
# ============================================================

def validar_carga_anterior(cursor, nombre_archivo):

    cursor.execute(
        f"""
        SELECT COUNT_BIG(*)
        FROM {TABLA_STAGING}
        WHERE AUDIT_DATE = ?
          AND SOURCE_FILE = ?
        """,
        AUDIT_DATE,
        nombre_archivo
    )

    cantidad = cursor.fetchone()[0]

    if cantidad > 0:

        raise RuntimeError(
            f"Ya existen {cantidad:,} registros para "
            f"{nombre_archivo} y auditoría {AUDIT_DATE}.\n"
            "Se cancela la carga para evitar duplicados."
        )

# ============================================================
# 7. PROCESAR UN REGISTRO
# ============================================================

def procesar_fila(fila, numero_fila, nombre_archivo, load_run_id):

    errores = []

    # Comprobar columnas adicionales o faltantes.
    if None in fila or any(valor is None for valor in fila.values()):

        return None, (
            load_run_id,
            numero_fila,
            nombre_archivo,
            AUDIT_DATE,
            "ESTRUCTURA",
            "La fila contiene columnas faltantes o adicionales.",
            json.dumps(fila, ensure_ascii=False, default=str)
        )

    try:

        brick_cod = convertir_entero(
            fila["BRICK_COD"],
            "BRICK_COD"
        )

    except (ValueError, InvalidOperation) as error:
        errores.append(("BRICK_COD", str(error)))

    packs = str(fila["PACKS"]).strip()

    if not packs:
        errores.append(("PACKS", "PACKS está vacío"))

    elif len(packs) > 50:
        errores.append(
            ("PACKS", "PACKS supera los 50 caracteres")
        )

    try:

        idperiodo = convertir_entero(
            fila["IDPERIODO"],
            "IDPERIODO"
        )

    except (ValueError, InvalidOperation) as error:
        errores.append(("IDPERIODO", str(error)))

    valores = {}

    for columna in [
        "SOLES",
        "UN",
        "USD",
        "UNIDAD_CUENTA"
    ]:

        try:

            valores[columna] = convertir_decimal(
                fila[columna]
            )

        except (ValueError, InvalidOperation) as error:

            errores.append((columna, str(error)))

    # Si hay errores, enviar el registro al log.
    if errores:

        columnas_error = ", ".join(
            columna for columna, mensaje in errores
        )

        mensajes = " | ".join(
            f"{columna}: {mensaje}"
            for columna, mensaje in errores
        )

        registro_error = (
            load_run_id,
            numero_fila,
            nombre_archivo,
            AUDIT_DATE,
            columnas_error[:100],
            mensajes[:1000],
            json.dumps(
                fila,
                ensure_ascii=False,
                default=str
            )
        )

        return None, registro_error

    # Registro listo para SQL Server.
    registro_valido = (
        load_run_id,
        numero_fila,
        nombre_archivo,
        AUDIT_DATE,
        brick_cod,
        packs,
        idperiodo,
        valores["SOLES"],
        valores["UN"],
        valores["USD"],
        valores["UNIDAD_CUENTA"]
    )

    return registro_valido, None


# ============================================================
# 8. INSERTAR LOTES
# ============================================================

def insertar_lote(
    cursor,
    conexion,
    registros_validos,
    registros_error
):

    if registros_validos:

        cursor.executemany(
            SQL_INSERT_STAGING,
            registros_validos
        )

    if registros_error:

        cursor.executemany(
            SQL_INSERT_ERRORES,
            registros_error
        )

    conexion.commit()


# ============================================================
# 9. CALCULAR USD CORPORATIVO
# ============================================================

def calcular_usd_corporativo(cursor, load_run_id):

    print("\nCalculando USD corporativo en SQL Server...")

    sql = f"""
    UPDATE S
    SET
        S.TIPO_CAMBIO_CORP = TC.TIPO_CAMBIO,
        S.USD_CORPORATIVO =
            CAST(
                S.SOLES / TC.TIPO_CAMBIO
                AS DECIMAL(30,19)
            )
    FROM {TABLA_STAGING} S
    INNER JOIN dbo.Config_TipoCambio TC
        ON YEAR(S.AUDIT_DATE) = TC.ANIO
    WHERE S.LOAD_RUN_ID = ?
    """

    cursor.execute(sql, load_run_id)

    cursor.connection.commit()

    cursor.execute(
        f"""
        SELECT
            COUNT_BIG(*) AS TOTAL,
            SUM(
                CASE
                    WHEN TIPO_CAMBIO_CORP IS NULL
                      OR USD_CORPORATIVO IS NULL
                    THEN 1
                    ELSE 0
                END
            ) AS PENDIENTES
        FROM {TABLA_STAGING}
        WHERE LOAD_RUN_ID = ?
        """,
        load_run_id
    )

    total, pendientes = cursor.fetchone()

    if pendientes:
        raise RuntimeError(
            f"Hay {pendientes:,} registros sin cálculo corporativo."
        )

    print(
        f"USD corporativo calculado correctamente: {total:,} registros."
    )

# 10. PROCESO PRINCIPAL
# ============================================================

def ejecutar_carga():

    ruta = Path(ARCHIVO)

    if not ruta.exists():
        raise FileNotFoundError(
            f"No existe el archivo: {ruta}"
        )

    nombre_archivo = ruta.name

    load_run_id = str(uuid.uuid4())

    total_leidos = 0
    total_insertados = 0
    total_rechazados = 0

    registros_validos = []
    registros_error = []

    print("\n" + "=" * 65)
    print("INICIO DE CARGA IQVIA DDD")
    print("=" * 65)

    print(f"Archivo: {nombre_archivo}")
    print(f"Auditoría: {AUDIT_DATE}")
    print(f"LOAD_RUN_ID: {load_run_id}")
    print(
        "Límite:",
        f"{LIMITE_REGISTROS:,}"
        if LIMITE_REGISTROS is not None
        else "Archivo completo"
    )

    conexion = None

    try:

        conexion = pyodbc.connect(
            CONNECTION_STRING,
            autocommit=False
        )

        cursor = conexion.cursor()

        cursor.fast_executemany = True

        # Verificar TC antes de leer el archivo.
        tipo_cambio = obtener_tipo_cambio(cursor)

        print(
            f"Tipo de cambio corporativo {AUDIT_DATE.year}: "
            f"{tipo_cambio}"
        )

        # Evitar duplicar la misma auditoría y archivo.
        validar_carga_anterior(
            cursor,
            nombre_archivo
        )

        # Validar encabezados.
        with open(
            ruta,
            "r",
            encoding="utf-8-sig",
            newline=""
        ) as archivo:

            lector = csv.DictReader(
                archivo,
                delimiter=","
            )

            if lector.fieldnames != COLUMNAS_ESPERADAS:

                raise ValueError(
                    "Los encabezados del TXT no coinciden.\n"
                    f"Encontrados: {lector.fieldnames}"
                )

            print("\nComenzando lectura e inserción...\n")

            for fila in lector:

                if (
                    LIMITE_REGISTROS is not None
                    and total_leidos >= LIMITE_REGISTROS
                ):
                    break

                total_leidos += 1

                # Primera fila del archivo = encabezado.
                numero_fila = total_leidos + 1

                registro, error = procesar_fila(
                    fila,
                    numero_fila,
                    nombre_archivo,
                    load_run_id
                )

                if registro is not None:

                    registros_validos.append(registro)

                if error is not None:

                    registros_error.append(error)

                # Insertar cuando se completa un lote.
                if len(registros_validos) + len(registros_error) >= TAMANO_LOTE:

                    insertar_lote(
                        cursor,
                        conexion,
                        registros_validos,
                        registros_error
                    )

                    total_insertados += len(registros_validos)
                    total_rechazados += len(registros_error)

                    registros_validos.clear()
                    registros_error.clear()

                    procesados = (
                        total_insertados + total_rechazados
                    )

                    print(
                        f"Procesados: {procesados:,} | "
                        f"Insertados: {total_insertados:,} | "
                        f"Rechazados: {total_rechazados:,}"
                    )

            # Insertar último lote pendiente.
            if registros_validos or registros_error:

                insertar_lote(
                    cursor,
                    conexion,
                    registros_validos,
                    registros_error
                )

                total_insertados += len(registros_validos)
                total_rechazados += len(registros_error)

                registros_validos.clear()
                registros_error.clear()

        # Aplicar TC corporativo a todos los registros de la ejecución.
        if total_insertados > 0:

            calcular_usd_corporativo(
                cursor,
                load_run_id
            )

        print("\n" + "=" * 65)
        print("RESUMEN DE CARGA")
        print("=" * 65)

        print(f"LOAD_RUN_ID: {load_run_id}")
        print(f"Registros leídos: {total_leidos:,}")
        print(f"Registros insertados: {total_insertados:,}")
        print(f"Registros rechazados: {total_rechazados:,}")

        print(
            f"Porcentaje de rechazo: "
            f"{(total_rechazados / total_leidos * 100):.6f}%"
            if total_leidos > 0
            else "Porcentaje de rechazo: 0%"
        )

        print("\nProceso finalizado.")

    except Exception:

        if conexion is not None:
            conexion.rollback()

        print("\nERROR DURANTE LA CARGA")
        print(
            "La ejecución puede contener lotes previamente confirmados."
        )
        print(f"LOAD_RUN_ID: {load_run_id}")

        raise

    finally:

        if conexion is not None:
            conexion.close()


# ============================================================
# 11. EJECUTAR
# ============================================================

if __name__ == "__main__":

    ejecutar_carga()
