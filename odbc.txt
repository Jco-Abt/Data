SELECT
    BRICK_COD,
    PACKS,
    IDPERIODO,
    COUNT_BIG(*) AS REPETICIONES
FROM dbo.Stg_DDD_Prueba_TC_DECIMAL
WHERE LOAD_RUN_ID = 'cee51a5c-c0b4-49dd-b736-dc187f19ba30'
GROUP BY BRICK_COD, PACKS, IDPERIODO
HAVING COUNT_BIG(*) > 1;

SELECT
    IDPERIODO,
    COUNT_BIG(*) AS REGISTROS
FROM dbo.Stg_DDD_Prueba_TC_DECIMAL
WHERE LOAD_RUN_ID = 'cee51a5c-c0b4-49dd-b736-dc187f19ba30'
GROUP BY IDPERIODO
ORDER BY IDPERIODO;

SELECT
    MIN(SOLES) AS MIN_SOLES,
    MAX(SOLES) AS MAX_SOLES,
    SUM(SOLES) AS TOTAL_SOLES,
    SUM(UN) AS TOTAL_UNIDADES,
    SUM(USD_IQVIA) AS TOTAL_USD_IQVIA,
    SUM(USD_CORPORATIVO) AS TOTAL_USD_CORPORATIVO
FROM dbo.Stg_DDD_Prueba_TC_DECIMAL
WHERE LOAD_RUN_ID = 'cee51a5c-c0b4-49dd-b736-dc187f19ba30';
