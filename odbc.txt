import time
import pyodbc

conn = pyodbc.connect(connection_string, timeout=30)
cursor = conn.cursor()

try:
    cursor.execute("""
        CREATE TABLE dbo.PruebaCarga100K (
            ID INT NOT NULL,
            DESCRIPCION VARCHAR(100)
        )
    """)

    cursor.fast_executemany = True

    total_registros = 100_000
    tamano_lote = 5_000

    inicio = time.perf_counter()

    for inicio_lote in range(1, total_registros + 1, tamano_lote):

        fin_lote = min(
            inicio_lote + tamano_lote,
            total_registros + 1
        )

        datos = [
            (i, f"Registro de prueba {i}")
            for i in range(inicio_lote, fin_lote)
        ]

        cursor.executemany("""
            INSERT INTO dbo.PruebaCarga100K
            (ID, DESCRIPCION)
            VALUES (?, ?)
        """, datos)

    conn.commit()

    tiempo = time.perf_counter() - inicio

    cursor.execute("""
        SELECT COUNT(*)
        FROM dbo.PruebaCarga100K
    """)

    cantidad = cursor.fetchone()[0]

    print("Registros insertados:", cantidad)
    print("Tiempo total:", round(tiempo, 2), "segundos")
    print(
        "Velocidad:",
        round(cantidad / tiempo, 0),
        "registros/segundo"
    )

except Exception as error:
    conn.rollback()
    print("Error:", error)

finally:
    cursor.close()
    conn.close()
