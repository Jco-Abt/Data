import pyodbc

conn = pyodbc.connect(connection_string)
cursor = conn.cursor()

try:
    cursor.execute("""
        CREATE TABLE dbo.PruebaCargaPython (
            ID INT NOT NULL,
            DESCRIPCION VARCHAR(100)
        )
    """)

    datos = [
        (i, f"Registro de prueba {i}")
        for i in range(1, 101)
    ]

    cursor.fast_executemany = True

    cursor.executemany("""
        INSERT INTO dbo.PruebaCargaPython
        (ID, DESCRIPCION)
        VALUES (?, ?)
    """, datos)

    conn.commit()

    cursor.execute("""
        SELECT COUNT(*)
        FROM dbo.PruebaCargaPython
    """)

    print("Registros insertados:", cursor.fetchone()[0])

except Exception as e:
    conn.rollback()
    print("Error:", e)

finally:
    cursor.close()
    conn.close()
