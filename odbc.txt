SELECT
    LOAD_RUN_ID,
    COUNT_BIG(*) AS TOTAL_INSERTADOS,
    MIN(SOURCE_ROW_NUMBER) AS PRIMERA_FILA,
    MAX(SOURCE_ROW_NUMBER) AS ULTIMA_FILA,
    COUNT_BIG(DISTINCT SOURCE_ROW_NUMBER) AS FILAS_DISTINTAS,
    MIN(LOADED_AT) AS INICIO_CARGA,
    MAX(LOADED_AT) AS ULTIMO_LOTE
FROM dbo.Stg_DDD
WHERE AUDIT_DATE = '2026-09-01'
  AND SOURCE_FILE = 'Dataset_ddd_202609.txt'
GROUP BY LOAD_RUN_ID;

SELECT
    LOAD_RUN_ID,
    COUNT_BIG(*) AS REGISTROS_RECHAZADOS
FROM dbo.Log_Errores_DDD
WHERE AUDIT_DATE = '2026-09-01'
  AND SOURCE_FILE = 'Dataset_ddd_202609.txt'
GROUP BY LOAD_RUN_ID;
