import pyodbc

SERVER = "NOMBRE_SERVIDOR"
DATABASE = "NOMBRE_BASE_DATOS"
USERNAME = "TU_USUARIO"

# Solicitar contraseña sin mostrarla en pantalla
from getpass import getpass
PASSWORD = getpass("Ingresa tu contraseña SQL: ")

connection_string = (
    "DRIVER={ODBC Driver 18 for SQL Server};"
    f"SERVER={SERVER};"
    f"DATABASE={DATABASE};"
    f"UID={USERNAME};"
    f"PWD={PASSWORD};"
    "Encrypt=yes;"
    "TrustServerCertificate=no;"
)

try:
    conn = pyodbc.connect(connection_string, timeout=15)

    cursor = conn.cursor()
    cursor.execute("""
        SELECT
            @@VERSION,
            DB_NAME(),
            SUSER_SNAME()
    """)

    result = cursor.fetchone()

    print("CONEXIÓN EXITOSA")
    print("Base de datos:", result[1])
    print("Usuario conectado:", result[2])

    conn.close()

except pyodbc.Error as error:
    print("ERROR DE CONEXIÓN:")
    print(error)
